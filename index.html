🚀 Quickstart (Vite / Browser)

        
import init, { compile_to_svg } from '@virasak/typst-math-wasm';
import wasmUrl from '@virasak/typst-math-wasm/wasm?url';

// 1. Initialize once with WASM URL
await init({ module_or_path: wasmUrl });

// 2. Compile math formula to SVG
const svg = compile_to_svg(`#set page(width: auto, height: auto, margin: 0.5em)
$ sum_(k=1)^n k = (n(n+1))/2 $`);

document.getElementById('math').innerHTML = svg;
